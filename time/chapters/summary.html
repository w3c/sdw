    Properties (organized by domain value)

    Domain :TemporalEntity

    • :hasBeginning rdfs:range :Instant .
    • :hasEnd rdfs:range :Instant .
    • :hasTemporalDuration rdfs:range :TemporalDuration .
    • :hasDuration rdfs:range :Duration .
    • :hasDurationDescription rdfs:range :GeneralDurationDescription .
    • :hasXSDDuration rdfs:range xsd:duration .
    • :before rdfs:range :TemporalEntity .
    • :after rdfs:range :TemporalEntity .

    Domain :Instant

    • :inTemporalPosition rdfs:range :TemporalPosition .
    • :inTimePosition rdfs:range :TimePosition .
    • :inDateTime rdfs:range :GeneralDateTimeDescription .
    • :inXSDDate rdfs:range xsd:date .
    • :inXSDDateTime (deprecated) rdfs:range xsd:dateTime .
    • :inXSDDateTimeStamp rdfs:range xsd:dateTimeStamp .
    • :inXSDgYearMonth rdfs:range xsd:gYearMonth .
    • :inXSDgYear rdfs:range xsd:gYear .

    Domain :Interval

    • :inside rdfs:range :Instant .

    Domain :ProperInterval

    • :intervalEquals rdfs:range :ProperInterval .
    • :intervalDisjoint rdfs:range :ProperInterval .
    • :intervalAfter rdfs:range :ProperInterval .
    • :intervalBefore rdfs:range :ProperInterval .
    • :intervalMeets rdfs:range :ProperInterval .
    • :intervalMetBy rdfs:range :ProperInterval .
    • :intervalOverlaps rdfs:range :ProperInterval .
    • :intervalOverlappedBy rdfs:range :ProperInterval .
    • :intervalStarts rdfs:range :ProperInterval .
    • :intervalStartedBy rdfs:range :ProperInterval .
    • :intervalFinishes rdfs:range :ProperInterval .
    • :intervalFinishedBy rdfs:range :ProperInterval .
    • :intervalContains rdfs:range :ProperInterval .
    • :intervalDuring rdfs:range :ProperInterval .
    • :intervalIn rdfs:range :ProperInterval .

    Domain :DateTimeInterval

    • :hasDateTimeDescription rdfs:range :GeneralDateTimeDescription .
    • :xsdDateTime (deprecated) rdfs:range xsd:dateTime .

    Domain :TemporalPosition or :GeneralDurationDescription

    • :hasTRS rdfs:range :TRS .

    Domain :TimePosition

    • :numericPosition rdfs:range xsd:decimal .
    • :nominalPosition rdfs:range xsd:string .

    Domain :GeneralDateTimeDescription

    • :timeZone rdfs:range :TimeZone (class) .

    • :dayOfYear rdfs:range xsd:nonNegativeInteger .
    • :dayOfWeek rdfs:range :DayOfWeek (class) .
    • :monthOfYear rdfs:range :MonthOfYear (class) .

    • :unitType rdfs:range :TemporalUnit .
